#include <stdio.h>
#include <stdlib.h>

struct Employee {
    int id;
    char name[30];
    float salary;
};

int main() {
    struct Employee emp;
    FILE *fp;
    int choice;

    while (1) {
        printf("\n1. Add Employee\n2. Display Employees\n3. Exit\nEnter choice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            fp = fopen("employee.txt", "ab");
            printf("Enter ID: ");
            scanf("%d", &emp.id);
            printf("Enter Name: ");
            scanf("%s", emp.name);
            printf("Enter Salary: ");
            scanf("%f", &emp.salary);
            fwrite(&emp, sizeof(emp), 1, fp);
            fclose(fp);
            printf("Record Added!\n");
        }

        else if (choice == 2) {
            fp = fopen("employee.txt", "rb");
            if (fp == NULL) {
                printf("No Records Found!\n");
                continue;
            }
            printf("\n--- Employee Records ---\n");
            while (fread(&emp, sizeof(emp), 1, fp))
                printf("ID: %d  Name: %s  Salary: %.2f\n", emp.id, emp.name, emp.salary);
            fclose(fp);
        }

        else if (choice == 3) {
            exit(0);
        }

        else {
            printf("Invalid choice!\n");
        }
    }
}
